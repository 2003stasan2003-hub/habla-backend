const express = require('express'); const http = require('http'); const { Server } = require('socket.io'); const app = express(); const server = http.createServer(app); const io = new Server(server); // Зберігаємо активних користувачів за їх socket.id const activeUsers = new Map(); io.on('connection', (socket) => { console.log(`Користувач підключився: ${socket.id}`); // Реєстрація користувача при вході socket.on('register_user', (userData) => { const username = userData && userData.name ? userData.name : 'Tester'; activeUsers.set(socket.id, { id: socket.id, name: username, status: 'online' }); // Оновлюємо список для всіх клієнтів broadcastUsersList(); }); // Обробка вхідних повідомлень / перекладів socket.on('chat_message', (data) => { const sender = activeUsers.get(socket.id); if (sender) { io.emit('chat_message', { senderId: socket.id, senderName: sender.name, text: data.text, targetLanguage: data.targetLanguage || 'en' }); } }); // Відключення користувача socket.on('disconnect', () => { console.log(`Користувач відключився: ${socket.id}`); activeUsers.delete(socket.id); broadcastUsersList(); }); }); function broadcastUsersList() { const usersArray = Array.from(activeUsers.values()); io.emit('users_update', usersArray); } const PORT = process.env.PORT || 3000; server.listen(PORT, () => { console.log(`Сервер запущено на порту ${PORT}`); });
